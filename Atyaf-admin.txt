    function openProduct(id){const p=id?db.products.find(x=>x.id===id):null;["pId","pName","pSku","pPrice","pDesc","pColors","pSizes","pImage"].forEach(x=>document.getElementById(x).value="");document.getElementById("pDiscount").value=0;document.getElementById("pStock").value=0;document.getElementById("pLow").value=5;document.getElementById("pSort").value=0;document.getElementById("pActive").checked=true;document.getElementById("pFeatured").checked=false;document.getElementById("pNew").checked=false;if(p){document.getElementById("pId").value=p.id;document.getElementById("pName").value=p.name;document.getElementById("pSku").value=p.sku;document.getElementById("pCategory").value=p.category;document.getElementById("pPrice").value=p.price;document.getElementById("pDiscount").value=p.discount;document.getElementById("pStock").value=p.stock;document.getElementById("pLow").value=p.low;document.getElementById("pSort").value=p.sort;document.getElementById("pDesc").value=p.desc;document.getElementById("pColors").value=p.colors;document.getElementById("pSizes").value=p.sizes;document.getElementById("pImage").value=p.image;document.getElementById("pActive").checked=p.active;document.getElementById("pFeatured").checked=p.featured;document.getElementById("pNew").checked=p.isNew}document.getElementById("productModalTitle").textContent=p?"تعديل المنتج":"إضافة منتج";document.getElementById("productModal").classList.add("show")}
    function closeProduct(){document.getElementById("productModal").classList.remove("show")}
    function saveProduct(){const id=+document.getElementById("pId").value||Date.now(),name=document.getElementById("pName").value.trim();if(!name){toast("اكتب اسم المنتج");return}const p={id,name,sku:document.getElementById("pSku").value.trim(),category:document.getElementById("pCategory").value,price:+document.getElementById("pPrice").value||0,discount:+document.getElementById("pDiscount").value||0,stock:+document.getElementById("pStock").value||0,low:+document.getElementById("pLow").value||5,sort:+document.getElementById("pSort").value||0,desc:document.getElementById("pDesc").value,colors:document.getElementById("pColors").value,sizes:document.getElementById("pSizes").value,image:document.getElementById("pImage").value,active:document.getElementById("pActive").checked,featured:document.getElementById("pFeatured").checked,isNew:document.getElementById("pNew").checked};const n=db.products.findIndex(x=>x.id===id);n>=0?db.products[n]=p:db.products.push(p);persist();renderProducts();closeProduct();toast("تم حفظ المنتج")}
    function renderProducts(){const q=(document.getElementById("productSearch")?.value||"").toLowerCase(),f=document.getElementById("productFilter")?.value||"";const a=db.products.filter(p=>(!q||p.name.toLowerCase().includes(q)||p.sku.toLowerCase().includes(q))&&(!f||p.category===f));document.getElementById("productRows").innerHTML=a.length?a.sort((x,y)=>x.sort-y.sort).map(p=>`<tr><td>${p.name}</td><td>${p.sku||"-"}</td><td>${p.category||"-"}</td><td>${money(p.price)}</td><td>${p.stock}</td><td><span class="pill">${p.active?"نشط":"موقوف"}</span></td><td><button class="secondary" onclick="openProduct(${p.id})">تعديل</button> <button class="danger" onclick="deleteProduct(${p.id})">حذف</button></td></tr>`).join(""):'<tr><td colspan="7" class="empty">لا توجد منتجات</td></tr>'}
    function deleteProduct(id){if(confirm("حذف المنتج؟")){db.products=db.products.filter(p=>p.id!==id);persist();renderProducts()}}

    function changeOrder(id,s){const o=db.orders.find(x=>x.id==id);if(o){o.status=s;persist();toast("تم تحديث حالة الطلب")}}

    function addCoupon(){const c=document.getElementById("couponCode").value.trim().toUpperCase(),p=+document.getElementById("couponPercent").value;if(!c||!p)return;db.coupons.push({code:c,percent:p});persist();renderCoupons();document.getElementById("couponCode").value="";document.getElementById("couponPercent").value=""}
